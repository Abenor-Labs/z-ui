import React, { useState, useRef, useEffect, useCallback } from "react";
import { Liquid } from "liquid-gooey";

// ════════════════════════════════════════════════════════════════
// Components built on liquid-gooey
//
//   npm install liquid-gooey
//
// The library gives you two primitives:
//   <Liquid type="morph">  — touching Items fuse, reshape, dissolve
//   <Liquid type="move">   — a moving Item drags a liquid tail
//   <Liquid.Item>          — anything inside that participates
//
// Everything below is composition on top of those two. The pattern
// throughout: animate Item *position*, let the goo do the rest.
// Keep text and fine detail OUTSIDE the Liquid container and layer
// it on top — the filter will smear it otherwise.
// ════════════════════════════════════════════════════════════════

function useReducedMotion() {
  const [reduced, setReduced] = useState(false);
  useEffect(() => {
    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
    setReduced(mq.matches);
    const on = (e) => setReduced(e.matches);
    mq.addEventListener("change", on);
    return () => mq.removeEventListener("change", on);
  }, []);
  return reduced;
}

// ---------------------------------------------------------------
// 1. GooeyFab — plus button that splits into a fan of actions
//    The classic. Actions start stacked under the trigger, so they
//    read as one blob, then separate as they travel outward.
// ---------------------------------------------------------------
export function GooeyFab({ actions = [], radius = 92, blur = 10, contrast = 20 }) {
  const [open, setOpen] = useState(false);
  const reduced = useReducedMotion();
  const spread = 90; // degrees of arc
  const start = 180; // pointing left-ish

  return (
    <div className="gfab" data-open={open}>
      <Liquid type="morph" blur={reduced ? 0 : blur} contrast={contrast}>
        {actions.map((a, i) => {
          const step = actions.length > 1 ? spread / (actions.length - 1) : 0;
          const angle = ((start + i * step) * Math.PI) / 180;
          const x = open ? Math.cos(angle) * radius : 0;
          const y = open ? Math.sin(angle) * radius : 0;
          return (
            <Liquid.Item key={a.label}>
              <span
                className="gfab-action"
                style={{
                  transform: `translate(${x}px, ${y}px) scale(${open ? 1 : 0.4})`,
                  transitionDelay: `${open ? i * 35 : (actions.length - i) * 25}ms`,
                }}
              />
            </Liquid.Item>
          );
        })}
        <Liquid.Item>
          <span className="gfab-trigger-blob" />
        </Liquid.Item>
      </Liquid>

      {/* icons ride above the goo layer, undistorted */}
      <div className="gfab-icons" aria-hidden={!open}>
        {actions.map((a, i) => {
          const step = actions.length > 1 ? spread / (actions.length - 1) : 0;
          const angle = ((start + i * step) * Math.PI) / 180;
          const x = open ? Math.cos(angle) * radius : 0;
          const y = open ? Math.sin(angle) * radius : 0;
          return (
            <button
              key={a.label}
              className="gfab-icon-btn"
              title={a.label}
              onClick={a.onClick}
              tabIndex={open ? 0 : -1}
              style={{
                transform: `translate(${x}px, ${y}px)`,
                opacity: open ? 1 : 0,
                transitionDelay: `${open ? i * 35 + 60 : 0}ms`,
              }}
            >
              {a.icon}
            </button>
          );
        })}
      </div>

      <button
        className="gfab-trigger"
        aria-expanded={open}
        aria-label={open ? "Close actions" : "Open actions"}
        onClick={() => setOpen((o) => !o)}
      >
        <span className="gfab-plus" data-open={open}>+</span>
      </button>
    </div>
  );
}

// ---------------------------------------------------------------
// 2. GooeyTabs — the active pill detaches, stretches across, and
//    re-forms on the new tab. Move type gives it the rubber trail.
// ---------------------------------------------------------------
export function GooeyTabs({ tabs = [], value, onChange, blur = 8, contrast = 18 }) {
  const [rects, setRects] = useState([]);
  const listRef = useRef(null);
  const reduced = useReducedMotion();

  const measure = useCallback(() => {
    if (!listRef.current) return;
    const parent = listRef.current.getBoundingClientRect();
    const next = Array.from(listRef.current.children).map((el) => {
      const r = el.getBoundingClientRect();
      return { left: r.left - parent.left, width: r.width };
    });
    setRects(next);
  }, []);

  useEffect(() => {
    measure();
    const ro = new ResizeObserver(measure);
    if (listRef.current) ro.observe(listRef.current);
    return () => ro.disconnect();
  }, [measure, tabs.length]);

  const activeIndex = tabs.findIndex((t) => t.value === value);
  const active = rects[activeIndex];

  return (
    <div className="gtabs">
      <Liquid type="move" blur={reduced ? 0 : blur} contrast={contrast}>
        <Liquid.Item>
          <span
            className="gtabs-pill"
            style={{
              transform: `translateX(${active?.left ?? 0}px)`,
              width: active?.width ?? 0,
            }}
          />
        </Liquid.Item>
      </Liquid>

      <div className="gtabs-list" ref={listRef} role="tablist">
        {tabs.map((t) => (
          <button
            key={t.value}
            role="tab"
            aria-selected={t.value === value}
            className="gtabs-tab"
            onClick={() => onChange(t.value)}
          >
            {t.label}
          </button>
        ))}
      </div>
    </div>
  );
}

// ---------------------------------------------------------------
// 3. GooeyToggle — handle stretches out of the track and snaps back.
//    Track and handle are both Items, so they stay fused mid-travel.
// ---------------------------------------------------------------
export function GooeyToggle({ checked, onChange, label, blur = 7, contrast = 22 }) {
  const reduced = useReducedMotion();
  return (
    <label className="gtoggle">
      <input
        type="checkbox"
        className="gtoggle-input"
        checked={checked}
        onChange={(e) => onChange(e.target.checked)}
      />
      <span className="gtoggle-visual" data-on={checked}>
        <Liquid type="morph" blur={reduced ? 0 : blur} contrast={contrast}>
          <Liquid.Item>
            <span className="gtoggle-track" />
          </Liquid.Item>
          <Liquid.Item>
            <span
              className="gtoggle-handle"
              style={{ transform: `translateX(${checked ? 26 : 0}px)` }}
            />
          </Liquid.Item>
        </Liquid>
      </span>
      {label && <span className="gtoggle-label">{label}</span>}
    </label>
  );
}

// ---------------------------------------------------------------
// 4. GooeyDock — neighbours bulge toward the hovered icon and fuse
//    with it, then release. Distance falloff drives the scale.
// ---------------------------------------------------------------
export function GooeyDock({ items = [], blur = 12, contrast = 20 }) {
  const [hover, setHover] = useState(null);
  const reduced = useReducedMotion();

  const scaleFor = (i) => {
    if (hover === null) return 1;
    const d = Math.abs(hover - i);
    if (d === 0) return 1.35;
    if (d === 1) return 1.15;
    if (d === 2) return 1.05;
    return 1;
  };
  const liftFor = (i) => {
    if (hover === null) return 0;
    const d = Math.abs(hover - i);
    return d === 0 ? -14 : d === 1 ? -7 : d === 2 ? -2 : 0;
  };

  return (
    <div className="gdock" onMouseLeave={() => setHover(null)}>
      <Liquid type="morph" blur={reduced ? 0 : blur} contrast={contrast}>
        {items.map((it, i) => (
          <Liquid.Item key={it.label}>
            <span
              className="gdock-blob"
              style={{
                transform: `translateY(${liftFor(i)}px) scale(${scaleFor(i)})`,
              }}
            />
          </Liquid.Item>
        ))}
      </Liquid>

      <div className="gdock-icons">
        {items.map((it, i) => (
          <button
            key={it.label}
            className="gdock-btn"
            title={it.label}
            onMouseEnter={() => setHover(i)}
            onFocus={() => setHover(i)}
            onBlur={() => setHover(null)}
            onClick={it.onClick}
            style={{
              transform: `translateY(${liftFor(i)}px) scale(${scaleFor(i)})`,
            }}
          >
            {it.icon}
          </button>
        ))}
      </div>
    </div>
  );
}

// ---------------------------------------------------------------
// 5. GooeyLoader — orbiting dots that repeatedly merge at the centre.
//    Pure decoration, no interaction, cheap to leave mounted.
// ---------------------------------------------------------------
export function GooeyLoader({ size = 64, dots = 3, blur = 6, contrast = 24 }) {
  const reduced = useReducedMotion();
  return (
    <span className="gloader" style={{ width: size, height: size }}>
      <Liquid type="morph" blur={reduced ? 0 : blur} contrast={contrast}>
        {Array.from({ length: dots }).map((_, i) => (
          <Liquid.Item key={i}>
            <span
              className="gloader-dot"
              style={{ animationDelay: `${(i * 1.2) / dots}s` }}
            />
          </Liquid.Item>
        ))}
      </Liquid>
    </span>
  );
}

// ---------------------------------------------------------------
// 6. GooeyCursor — Move type following the pointer inside a region.
//    Good for hero sections; scope it, don't attach to the document.
// ---------------------------------------------------------------
export function GooeyCursor({ children, blur = 14, contrast = 18 }) {
  const ref = useRef(null);
  const [pos, setPos] = useState({ x: -999, y: -999 });
  const reduced = useReducedMotion();

  const onMove = (e) => {
    const r = ref.current.getBoundingClientRect();
    setPos({ x: e.clientX - r.left, y: e.clientY - r.top });
  };

  return (
    <div
      className="gcursor"
      ref={ref}
      onPointerMove={onMove}
      onPointerLeave={() => setPos({ x: -999, y: -999 })}
    >
      <Liquid type="move" blur={reduced ? 0 : blur} contrast={contrast}>
        <Liquid.Item>
          <span
            className="gcursor-blob"
            style={{ transform: `translate(${pos.x - 30}px, ${pos.y - 30}px)` }}
          />
        </Liquid.Item>
      </Liquid>
      <div className="gcursor-content">{children}</div>
    </div>
  );
}
